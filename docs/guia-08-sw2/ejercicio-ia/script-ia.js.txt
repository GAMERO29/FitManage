import http from 'k6/http';
import { check, sleep } from 'k6';

// Configuración de la prueba de carga
export const options = {
  stages: [
    { duration: '30s', target: 200 },   // calentamiento
    { duration: '1m', target: 1000 },   // subida a 1000 usuarios
    { duration: '3m', target: 1000 },   // carga sostenida
    { duration: '1m', target: 5000 },   // pico de estrés
    { duration: '30s', target: 0 },     // bajada
  ],
  thresholds: {
    http_req_duration: ['p(95)<500'],
  },
};

const BASE_URL = 'https://api.fitmanage.com';

export default function () {
  // 1. Iniciar sesión para obtener el token JWT
  const login = http.post(`${BASE_URL}/api/v1/auth/login`, JSON.stringify({
    username: 'admin',
    password: 'admin123',
  }), { headers: { 'Content-Type': 'application/json' } });

  const token = login.json('access_token');

  // 2. Consultar el estado de un socio por cédula
  const cedula = Math.floor(Math.random() * 100000000);
  const res = http.get(`${BASE_URL}/api/v1/socios/${cedula}/estado`, {
    headers: { Authorization: `Bearer ${token}` },
  });

  check(res, {
    'status es 200': (r) => r.status === 200,
  });
}
